function path_to_root(parents, current_vertex): # parents is the dictionary that forms the tree, current_vertex is the vertex we want to know the path of
    path <- []  # initialization of path
    while current_vertex is not None:   # loop until we pass the root of the tree
        path <- path + [current_vertex] # add current_vertex to the path
        current_vertex <- parents[current_vertex]   # we move current_vertex to its parent
    return path


# parents is the dictionary that forms the tree, current_vertex is the vertex we were testing and cycle_vertex is the one found twice in the sub-graph
function reconstitute_cycle(parents, current_vertex, cycle_vertex):
    cycle_path <- path_to_root(parents, cycle_vertex)   # recover the path from cycle_vertex to the root of the tree
    path_current <- path_to_root(parents, current_vertex)   # recover the path from current_vertex to the root of the tree
    i <- sizeof(cycle_path) - 1 # index we are going to use for cycle_path initialized at the end of cycle_path
    j <- sizeof(path_current) - 1   # index we are going to use for path_current initialized at the end of path_current
    while i >= 0 and j >= 0 and cycle_path[i] = path_current[j]:    # we loop until cycle_path and path_current diverge
        i <- i - 1  # we decrease the indexes at each loop to make them go from the root
        j <- j - 1
    # we recreate the cycle by going from cycle_vertex to the 1st parent in common then we go to current_vertex and add cycle_vertex again at the end to close the cycle
    return cycle_path[to i + 1] + path_current[to j + 1 from the end] + [cycle_vertex]


function search_cycle(additional_edges):    # additional_edges here can come from when a 0 should be consider as an edge for connectivity
    visited_global <- set() # vertices that have already been visited
    cycle <- [] # potential cycle
    while sizeof(visited_global) < nb_suppliers + nb_customers and cycle is empty:  # we loop until we already visited all the vertices or a cycle is found
        initial_vertex <- (0, 0)    # the vertex from which we will start the search, the first 0 is for supplier, the second is for it's number (0 indexing)
        while initial_vertex[1] < nb_suppliers - 1 and initial_vertex is in visited_global: # we search the first vertex to not have been visited starting with suppliers
            initial_vertex <- (0, initial_vertex[1] + 1)
        if initial_vertex is in visited_global:
            initial_vertex <- (1, 0)
            while initial_vertex[1] < nb_customers - 1 and initial_vertex is in visited_global:
                initial_vertex <- (1, initial_vertex[1] + 1)
        parents <- {initial_vertex: None}   # dictionary that represent the tree, since the graph is undirected, each vertex can only have 1 parent
        queue <- [initial_vertex]   # queue for BFS
        head <- 0   # we used a int for the queue head but we could have used deque
        visited <- {queue[head]}    # the vertex already visited in the sub-graph which is not really necessary other than for quality of life
        while head < sizeof(queue) and cycle is empty:  # we loop until we emptied the queue or a cycle is found
            current_vertex <- queue[head]   # quality of life current_vertex
            if current_vertex[0] = 0:   # if current_vertex is a supplier
                customer_nb <- 0
                while customer_nb < nb_customers and cycle is empty:    # we loop until we checked for every potential edges coming from current_vertex or a cycle is found
                    if ((transport_proposal_matrix[current_vertex[1]][customer_nb] != 0 or [current_vertex[1], customer_nb] is in additional_edges)
                            and (1, customer_nb) != parents.get(current_vertex)):   # check whether an edge is part of the sub-graph and that it is not just its parent
                        if (1, customer_nb) is in visited:  # if the vertex is already visited in the sub-graph then there is a cycle that goes trough it and current_vertex
                            cycle <- reconstitute_cycle(parents, current_vertex, (1, customer_nb))  # recover the cycle
                        else:
                            queue <- queue + [(1, customer_nb)] # we add the new vertex to the queue, put it in visited and put current_vertex as his parent
                            visited <- visited + {(1, customer_nb)}
                            parents[(1, customer_nb)] <- current_vertex
                    customer_nb <- customer_nb - 1
            else:   # if current_vertex is a customer
                # same as before but with every customer things replace by supplier things and vice versa
            head <- head + 1
        if initial_vertex = (0, 0): # check if the initial_vertex for the sub-graph is P1
            connected <- visited    # the vertices visited in this sub-graph will be our base to connect the rest
        visited_global <- visited_global + visited  # we add the locally visited vertices inside visited_global
    return cycle, connected


function supress_cycle(cycle):
    delta <- infinity   # we initialize delta at infinity
    for vertex_nb from 0 to sizeof(cycle) - 2:  # here we consider the edges of the cycle so there is 1 less than vertices
        if vertex_nb is odd:    # check if we will subtract delta for this edge, by default we put that the first edge adds delta so if the index is odd it is a subtraction
            current <- cycle[vertex_nb] # start of the edge
            next_vertex <- cycle[vertex_nb + 1] # end of the edge
            if current[0] = 0:  # if the start of the edge is a supplier
                delta <- min(delta, transport_proposal_matrix[current[1]][next_vertex[1]])  # the minimum of the ones that will be negative before applying the sign
            else:
                # same but with the supplier and customer part inverted

    # in the actual code the edges that have been removed are kept in memory, however this is only for display purposes
    for vertex_nb from 0 to sizeof(cycle) - 2:  # we do the for loop 2 times in order to have the best delta before starting changing values
        current <- cycle[vertex_nb]
        next_vertex <- cycle[vertex_nb + 1]
        if vertex_nb is even:   # check if we will add delta for this edge
            if current[0] = 0:  # check if the start of the edge is a supplier
                transport_proposal_matrix[current[1]][next_vertex[1]] <- transport_proposal_matrix[current[1]][next_vertex[1]] + delta  # we add delta for this edge
            else:
                # same but with the supplier and customer part inverted
        else:
            if current[0] = 0:  # check if the start of the edge is a supplier
                transport_proposal_matrix[current[1]][next_vertex[1]] <- transport_proposal_matrix[current[1]][next_vertex[1]] - delta  # we subtract delta for this edge
            else:
                # same but with the supplier and customer part inverted

function connect_graph(visited, not_connected): # visited is the main sub-graph, not connected are the vertices not in visited
    lowest_cost <- infinity # we set the lowest cost to infinity
    vertex_nb <- 0  # index for not_connected
    added_edge <- []    # the edge we will add into the transport proposal
    new_visited <- visited  # the new sub-graph for the added edge, we keep it separate because we have to test multiple edges so we have to keep the original sub-graph
    while vertex_nb < sizeof(not_connected):    # we try for each vertex in not_connected
        current <- not_connected[vertex_nb] # quality of life current inside not_connected
        if current[0] = 0:  # check if current is a supplier
            for customer_nb from nb_customers - 1:  # test each potential edges
                if (1, customer_nb) is in visited and costs_matrix[current[1]][customer_nb] < lowest_cost:  # check that it connect and that the edge is better
                    transport_proposal_matrix[current[1]][customer_nb] <- 1 # we put a dummy 1 into the transport proposal for the edge to test if it create a cycle
                    cycle_test, visited_test <- search_cycle([])    # we recover the cycle that would be created if any and the potential new sub-graph
                    if cycle_test is empty: # check if no cycle is created, if it the case this is our new best way to connect a vertex of not_connected
                        lowest_cost <- costs_matrix[current[1]][customer_nb]
                        added_edge <- [current[1], customer_nb]
                        new_visited <- visited_test
                    transport_proposal_matrix[current[1]][customer_nb] <- 0 # we remove the dummy 1 of the transport proposal to test the other potential edges
        else:
            # same but with supplier instead of customer, we always keep the supplier as the first component of the added edge and the customer as the second one
        vertex_nb <- vertex_nb + 1
    # we put 1 in the transport proposal for the newly added edge for it to be retained when we will be connecting the rest of not_connected
    transport_proposal_matrix[added_edge[0]][added_edge[1]] <- 1
    return added_edge, new_visited


function test_degenerate():
    cycle <- [initialization]   # we initialize cycle with a value to enter the while since there is no do-while in python
    while cycle is not empty:   # since we update the value of cycle at the beginning of the while it will always do an extra loop, here it only helps for the display however
        cycle, visited <- search_cycle([])  # we search for cycles inside the proposal and recuperate vertices connected to P1
        if cycle is not empty:
            supress_cycle(cycle)    # supress the cycle, in the actual code we return delta and the suppressed edges here however it is only for display reasons

    not_connected <- [initialization]   # we initialize not_connected with a value to enter the while since there is no do-while in python
    additional_edges <- []  # the edges that will be taken into account for connectivity
    while not_connected is not empty:   # since we update the value of not_connected at the beginning of the while it will always do an extra loop here it is used for display and to put back the values of the transport proposal
        not_connected <- test_connectivity(visited) # get the vertices that are not connected to the sub-graph that has P1 as its root
        if not_connected is not empty:  # check if there are vertices that are not connected to the graph
            added_edge, visited <- connect_graph(visited, not_connected)    # here it connect at least one vertex, we recover the additional edges to consider and the new sub-graph
            additional_edges <- additional_edges + [added_edge] # we put the new edge in our list of additional edges
        else:   # if every vertices are in the main sub-graph, the graph is connected
            for each edge_added in additional_edges:
                transport_proposal_matrix[edge_added[0]][edge_added[1]] <- 0    # we can now put back the 0s in our transport proposal while still keeping in memory the additional edges

    return additional_edges


function compute_potential_marginal_costs(additional_edges):    # additional_edges are edges that should be taken into account even if they are equal to 0 for connectivity
    parents <- {(0, 0): None}   # we initialize all the necessary and quality of life things for BFS
    queue <- [(0, 0)]   # we initialize our queue for the BFS with P1
    head <- 0
    visited <- {queue[head]}
    vertex_potentials <- [[0]*nb_suppliers, [0]*nb_customers]   # we initialize them at 0 to access them by index, the potential of the vertex P1 will always be 0
    while head < sizeof(queue): # here we start our BFS
        current_vertex <- queue[head]   # quality of life current_vertex
        if current_vertex[0] = 0:   # check if current_vertex is a supplier
            for customer_nb from 0 to nb_customers - 1: # we test for every edges
                if ((transport_proposal_matrix[current_vertex[1]][customer_nb] != 0 or [current_vertex[1], customer_nb] is in additional_edges)
                        and (1, customer_nb) != parents[current_vertex]):   #  check whether an edge is part of the sub-graph and that it is not just its parent
                    queue <- queue + [(1, customer_nb)] # usual BFS stuffs
                    visited <- visited + {(1, customer_nb)}
                    parents[(1, customer_nb)] <- current_vertex
                    potential_costs_matrix[current_vertex[1]][customer_nb] <- costs_matrix[current_vertex[1]][customer_nb]  # we put the potential cost as the actual cost
                    # we compute the potential for the vertex at the end point of the edge
                    vertex_potentials[1][customer_nb] <- -potential_costs_matrix[current_vertex[1]][customer_nb] + vertex_potentials[0][current_vertex[1]]
        else:
            # same but with supplier and customer stuffs inverted and also there is no - when computing vertex potentials
        head <- head + 1

    for supplier_nb from 0 to nb_suppliers - 1: # we compute the potential and marginal costs for every edge
        for customer_nb from 0 to nb_customers - 1: # in the submitted code there was a mistake here where we would only compute them when potential costs was 0
            potential_costs_matrix[supplier_nb][customer_nb] <- vertex_potentials[0][supplier_nb] - vertex_potentials[1][customer_nb]
            marginal_costs_matrix[supplier_nb][customer_nb] <- costs_matrix[supplier_nb][customer_nb] - potential_costs_matrix[supplier_nb][customer_nb]


function test_optimal():
    lowest_marginal_cost <- 0   # initialize at 0 to only consider negative marginal costs
    new_edge <- []  # the edge that needs to be added if the proposition is not optimal
    for supplier_nb from 0 to nb_suppliers - 1: # we check for every edge
        for customer_nb from 0 to nb_customers - 1:
            if marginal_costs_matrix[supplier_nb][customer_nb] < lowest_marginal_cost:  # check if the marginal cost for the edge is lower than our current lowest cost
                lowest_marginal_cost <- marginal_costs_matrix[supplier_nb][customer_nb] # we put our new lowest in memory
                new_edge <- [supplier_nb, customer_nb]
    return new_edge

function stepping_stone():
    additional_edges <- test_degenerate()   # we test if the proposal is degenerate and recover the additional edges
    compute_potential_marginal_costs(additional_edges)  # we compute the potential and marginal costs of the transport proposal
    new_edge <- test_optimal()  # we test whether the proposal is optimal and in the case it is not, we recover the first edge that needs to be added

    while new_edge is not empty:    # we loop until the proposal is optimal
        additional_edges <- additional_edges + [new_edge]   # we put new_edge in additional_edges to take them into account latter
        cycle, visited <- search_cycle(additional_edges)    # we reconstitute the cycle using the same function as before, this time however we put additional_edges to be taken into account in the transport proposal, visited is useless here
        removed_edges <- add_improving_edge(cycle, new_edge)    # we supress the cycle and add the proper value for the new edge in the transport proposal, we also retrieve the removed edges

        leaving_edge <- removed_edges[0]    # the edge that will actually be leaving the transport proposal

        for each removed_edge in removed_edges: # we go through every edge which values have fell to 0
            edge_formatted <- [removed_edge[0][1], removed_edge[1][1]]  # we reformat the edge to coincide with the format used for additional_edges
            if removed_edge = leaving_edge: # check if the edge is the one that should be leaving
                if edge_formatted in additional_edges:  # if the edge is present in additional_edges it should be removed
                    additional_edges <- additional_edges - [edge_formatted]
            else:
                if edge_formatted not in additional_edges:  # if the edge is not present in additional_edges it should be added for the next loop
                    additional_edges <- additional_edges + [edge_formatted]

        compute_potential_marginal_costs(additional_edges)  # we update the potential and marginal costs
        new_edge <- test_optimal()  # we test if the transport proposal is now optimal or if another step of the stepping stone should be computed
